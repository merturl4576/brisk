- Commits: plain conventional subjects; every message ends with `Co-Authored-By: Claude Opus 5.5 <noreply@anthropic.com>`.
- `TreatWarningsAsErrors` is on in every project.
- Copy rules: titles ≤ ~6 words, cost lines ≤ 6 words, no paragraphs on screen.
- Colours: background radial #0f3549 → #08222f (42 %) → #050d14; accent #5fd4e8; green #3ecf8e; amber #e8b45f; red #e8705f; text #e6f4f8; muted #7fa3b1; unlit tick #1a3a48.

- Motion budget: Home visible and idle ≤ 3 % total CPU (10 s average); minimized/hidden → ambient clocks stopped. Ambient motion is gated by `Settings.Motion` + window visibility, never by Windows' animation flag.

## Pre-plan finding (profiling, 2026-10-16)

Spec Engine change 1 asked the first task to profile the 45 s plan. Done while writing this plan with a read-only harness (scratchpad `profiler/`): warm, the full parallel scan takes **3.0 s** (2.3 s without the lock probe) and `BuildPlan` **2.5 s** (delivery-optimization detect 1.3 s, vm-platform 0.9 s, recycle-bin size 1.0 s); `brisk run` re-timed at **6.5 s**. The 44.6 s was a cold file cache (first walk of npm-cache and user-temp since boot). **Ruling:** no parallelization work — the engine is not the bottleneck; the progressive Looking state (privacy+speed first, space climbing per target) is the answer for a cold first launch. Success criterion 1 is measured warm (≤ 20 s; expected ≈ 6 s) and, cold, the screen must show progress throughout. Cost if wrong: a cold first launch on a slow HDD shows a climbing number for longer than 20 s.

    /// The rows that need no cleanup scan: they arrive first in the GUI.
    public IReadOnlyList<ActionGroup> BuildFast() => new[]
        {
            new ActionGroup(ActionGroupKind.Speed, SpeedRows().ToList()),
            new ActionGroup(ActionGroupKind.Privacy, PrivacyRows().ToList()),
        }.Where(g => g.Items.Count > 0).ToList();

    public ActionGroup? BuildSpace(ScanResult scan)
    {
        var rows = SpaceRows(scan).ToList();
        return rows.Count == 0 ? null : new ActionGroup(ActionGroupKind.Space, rows);
    }

    public static ActionPlan Compose(DateTime nowUtc, ActionGroup? space, IReadOnlyList<ActionGroup> fast,
        ScanResult scan)
    {
        var groups = new List<ActionGroup>();
        if (space is not null) groups.Add(space);
        groups.AddRange(fast.Where(g => g.Kind == ActionGroupKind.Speed));
        groups.AddRange(fast.Where(g => g.Kind == ActionGroupKind.Privacy));
        return new ActionPlan(nowUtc, groups, scan);
    }

    public ActionPlan Build(ScanResult scan, DateTime nowUtc) => Compose(nowUtc, BuildSpace(scan), BuildFast(), scan);

namespace BriskEngine.Actions;

public enum ResultTone { Good, Warn, Muted, Bad }

public sealed record ResultWord(string Key, string English, string? Reason, ResultTone Tone);

/// The word a finished row gets — the same for the CLI and the window.
public static class ActionResultWords
{
    public static ResultWord For(ActionGroupKind? group, RowResult r)
    {
        ResultWord W(string key, ResultTone tone, string? reason = null) => new(key, ActionStrings.En(key), reason, tone);
        return r.Outcome switch
        {
            RowOutcome.Skipped when r.Reason == ActionRunner.NeedsAdmin => W("action.result.needs-admin", ResultTone.Muted),
            RowOutcome.Skipped => W("action.result.skipped", ResultTone.Muted, r.Reason),
            RowOutcome.Failed => W("action.result.failed", ResultTone.Bad, r.Reason),
            RowOutcome.PendingRestart => W("action.result.pending", ResultTone.Muted),
            _ => r.ReadBack switch
            {
                ReadBackState.Reverted => W("action.result.reverted", ResultTone.Warn),
                ReadBackState.WrittenButIgnored => W("action.result.ignored", ResultTone.Warn),
                ReadBackState.WrittenButUnverified => W("action.result.written", ResultTone.Good),
                _ => W(group == ActionGroupKind.Privacy ? "action.result.off" : "action.result.done", ResultTone.Good),
            },
        };
    }
}
```

    /// null when the command names no ".exe".
    internal static string? ExecutablePath(string? command)
    {
        if (string.IsNullOrWhiteSpace(command)) return null;
        var end = command.IndexOf(".exe", StringComparison.OrdinalIgnoreCase);
        if (end < 0) return null;
        return Environment.ExpandEnvironmentVariables(command[..(end + 4)].TrimStart('"', ' '));
    }

using System;
using System.Linq;
using System.Text.RegularExpressions;
using BriskEngine.Diagnostics;

namespace BriskEngine.Actions;

/// The name a startup row shows: the program's own description
/// ("Microsoft Edge"), else the entry's name without machine-made tails.
public static class StartupTitle
{
    private static readonly string[] Generic = { "Update", "Updater", "Launcher", "Helper", "Application", "Electron" };
    private static readonly Regex IdTail = new(
        @"[_ ]?(\{?[0-9A-Fa-f]{8}-[0-9A-Fa-f]{4}-[0-9A-Fa-f]{4}-[0-9A-Fa-f]{4}-[0-9A-Fa-f]{12}\}?|[0-9A-Fa-f]{8,})$");

    public static string For(StartupEntry e, Func<string, string?> describe)
    {
        if (StartupManager.ExecutablePath(e.Command) is { } path
            && describe(path) is { } d && !string.IsNullOrWhiteSpace(d)
            && !Generic.Contains(d.Trim(), StringComparer.OrdinalIgnoreCase))
            return d.Trim();
        return Clean(e.Name);
    }

    internal static string Clean(string name)
    {
        var n = name.EndsWith(".lnk", StringComparison.OrdinalIgnoreCase) ? name[..^4] : name;
        var cut = IdTail.Replace(n, "");
        return cut.Length == 0 ? n : cut;
    }
}
```

Write the regex as `@"[_ ](\{?GUID\}?|[0-9A-Fa-f]{8,})$"` with the GUID spelled as above, and run the test cases.

using System.Globalization;

namespace Brisk.Session;

/// Sizes as the window prints them: GB with one decimal under ten, whole
/// above; MB whole; in the window's language (26,4 → "26"; 5,1).
public static class Sizes
{
    private const double Gb = 1L << 30, Mb = 1L << 20;

    public static (string Number, string Unit) Split(long bytes, CultureInfo c)
    {
        if (bytes <= 0) return ("0", "GB");
        if (bytes >= 1L << 30)
        {
            var g = bytes / Gb;
            return (g < 10 ? g.ToString("0.0", c) : Math.Floor(g).ToString("0", c), "GB");
        }
        if (bytes >= 1L << 20) return (Math.Floor(bytes / Mb).ToString("0", c), "MB");
        return ("<1", "MB");
    }

    public static string Format(long bytes, CultureInfo c)
    {
        var (n, u) = Split(bytes, c);
        return $"{n} {u}";
    }
}

/// The space numbers while the scan is still sizing targets — what the hero

    public static (long Found, long Ticked) Estimate(IEnumerable<TargetScanResult> sized)
    {
        long found = 0, ticked = 0;
        foreach (var t in sized)
        {
            var bytes = t.SkippedReason is null ? t.ReclaimableBytes : t.TotalBytes;
            found += t.TotalBytes;
            var row = ActionCatalog.Rows.FirstOrDefault(r =>
                r.Group == ActionGroupKind.Space && r.LinkKind == ActionLinkKind.Cleanup && r.LinkIds.Contains(t.Target.Id));
            if (row is { DefaultOn: true }) ticked += bytes;
        }
        return (found, ticked);
    }

using System;
using System.Collections.Generic;
using System.Linq;
using System.Threading.Tasks;
using BriskEngine.Actions;
using BriskEngine.Models;

namespace Brisk.Session;

public enum SessionState { Looking, Ready, Running, Done, Failed }

/// The window's one state machine: the plan, the ticks, the run and its
/// receipt. No WPF — every change is applied through `post` (the
/// dispatcher in the app, a lock in tests) and announced by Changed.
public sealed class ActionSession
{
    /// Rows whose deletions cannot come back; they get the one confirmation.
    public static readonly IReadOnlySet<string> Unrecoverable =
        new HashSet<string> { "space.downloads", "space.recycle-bin", "space.windows-old" };

    private readonly IActionBackend _backend;
    private readonly Action<Action> _post;
    private readonly Dictionary<string, bool> _rowOverrides = new(StringComparer.OrdinalIgnoreCase);
    private readonly Dictionary<string, bool> _fileOverrides = new(StringComparer.OrdinalIgnoreCase);
    private readonly Dictionary<string, RowResult> _results = new(StringComparer.OrdinalIgnoreCase);
    private readonly List<TargetScanResult> _sized = new();
    private readonly object _gate = new();
    private bool _busy;

    public ActionSession(IActionBackend backend, Action<Action> post)
    {
        _backend = backend;
        _post = post;
    }

    public event Action? Changed;

    public SessionState State { get; private set; } = SessionState.Looking;
    public IReadOnlyList<ActionGroup> FastGroups { get; private set; } = Array.Empty<ActionGroup>();
    public ActionGroup? SpaceGroup { get; private set; }
    public bool SpaceReady { get; private set; }
    public ActionPlan? Plan { get; private set; }
    public IReadOnlyList<TargetScanResult> SizedSoFar => _sized;
    public int TargetCount { get; } = CleanupTargetRegistry.All.Count;
    public string? Error { get; private set; }
    public string? CurrentRowId { get; private set; }
    public IReadOnlyDictionary<string, RowResult> Results => _results;
    public ActionGroupKind? RunGroup { get; private set; }
    public RunRecord? LastRun { get; private set; }
    public bool RanThisSession { get; private set; }
    public Receipt? Receipt { get; private set; }
    public IReadOnlyList<RowResult>? UndoResults { get; private set; }
    public bool Busy => _busy;

    public IEnumerable<ActionItem> Items => Plan?.Items
        ?? (SpaceGroup?.Items ?? Enumerable.Empty<ActionItem>()).Concat(FastGroups.SelectMany(g => g.Items));

    public bool CanUndo => !_busy && LastRun is { UndoneUtc: null, UndoableRuleIds.Count: > 0 };

    // ---- ticks
    public bool IsTicked(string rowId) =>
        _rowOverrides.TryGetValue(rowId, out var on) ? on : Items.FirstOrDefault(i => i.Id == rowId)?.DefaultOn ?? false;

    public void SetTicked(string rowId, bool on) { _rowOverrides[rowId] = on; Changed?.Invoke(); }

    public bool IsFileTicked(string path) => !_fileOverrides.TryGetValue(path, out var on) || on;

    public void SetFileTicked(string path, bool on) { _fileOverrides[path] = on; Changed?.Invoke(); }

    /// Bytes a row would free as ticked now (the Downloads row: its ticked files).
    public long SelectedBytes(ActionItem item) => item.Files.Count > 0
        ? item.Files.Where(f => IsFileTicked(f.Path)).Sum(f => f.Bytes)
        : item.Bytes ?? 0;

    /// Ticked and able to do something: a Downloads row with no ticked file is not.
    public bool WillRun(ActionItem item) => IsTicked(item.Id) && (item.Files.Count == 0 || SelectedBytes(item) > 0);

    public ActionSelection Selection(ActionGroupKind? group) => new(
        Items.Where(i => (group is null || i.Group == group) && WillRun(i)).Select(i => i.Id)
            .ToHashSet(StringComparer.OrdinalIgnoreCase),
        Items.SelectMany(i => i.Files).Where(f => IsFileTicked(f.Path)).Select(f => f.Path)
            .ToHashSet(StringComparer.OrdinalIgnoreCase));

    public IReadOnlyList<ActionItem> NeedsConfirmation(ActionGroupKind? group) =>
        Items.Where(i => Unrecoverable.Contains(i.Id) && (group is null || i.Group == group) && WillRun(i)).ToList();

    // ---- operations (each refuses while another runs)
    private bool TryBegin()
    {
        lock (_gate) { if (_busy) return false; _busy = true; return true; }
    }

    private void End() { lock (_gate) _busy = false; }

    private void Apply(Action change) => _post(() => { change(); Changed?.Invoke(); });

    public Task LookAsync() => TryBegin() ? LookCore() : Task.CompletedTask;

    public Task RescanAsync() => LookAsync();

    private async Task LookCore()
    {
        Apply(() =>
        {
            State = SessionState.Looking; Error = null; Plan = null; SpaceGroup = null; SpaceReady = false;
            FastGroups = Array.Empty<ActionGroup>(); _sized.Clear(); _results.Clear(); CurrentRowId = null;
        });
        try
        {
            var fast = Task.Run(() =>
            {
                var groups = _backend.BuildFast();
                Apply(() => FastGroups = groups);
                return groups;
            });
            var space = Task.Run(() =>
            {
                var scan = _backend.ScanSpace(t => Apply(() => _sized.Add(t)));
                var group = _backend.BuildSpace(scan);
                Apply(() => { SpaceGroup = group; SpaceReady = true; });
                return (scan, group);
            });
            var previous = Task.Run(() =>
            {
                var run = _backend.LatestRun();
                return (run, receipt: run is null ? null : _backend.RefreshReceipt(run.RunId));
            });
            await Task.WhenAll(fast, space, previous).ConfigureAwait(false);
            var (scan, group) = space.Result;
            var plan = ActionPlanBuilder.Compose(DateTime.UtcNow, group, fast.Result, scan);
            Apply(() =>
            {
                Plan = plan;
                if (!RanThisSession) { LastRun = previous.Result.run; Receipt = previous.Result.receipt; }
                State = SessionState.Ready;
            });
        }
        catch (Exception ex)
        {
            var message = (ex as AggregateException)?.InnerException?.Message ?? ex.Message;
            Apply(() => { State = SessionState.Failed; Error = message; });
        }
        finally { End(); }
    }

    public Task RunAsync(ActionGroupKind? group, bool skipUnrecoverable = false)
    {
        if (State != SessionState.Ready || Plan is null || !TryBegin()) return Task.CompletedTask;
        return RunCore(group, skipUnrecoverable);
    }

    private async Task RunCore(ActionGroupKind? group, bool skipUnrecoverable)
    {
        if (skipUnrecoverable)
            foreach (var item in NeedsConfirmation(group)) _rowOverrides[item.Id] = false;
        var plan = Plan!;
        var selection = Selection(group);
        Apply(() => { State = SessionState.Running; RunGroup = group; _results.Clear(); UndoResults = null; });
        try
        {
            var (run, receipt) = await Task.Run(() =>
            {
                if (group is null or ActionGroupKind.Space) _backend.ProcessDeferred();
                var record = _backend.Run(plan, selection,
                    start => Apply(() => CurrentRowId = start.Id),
                    row => Apply(() => _results[row.RowId] = row));
                return (record, _backend.RefreshReceipt(record.RunId));
            }).ConfigureAwait(false);
            Apply(() =>
            {
                LastRun = run; Receipt = receipt; RanThisSession = true;
                CurrentRowId = null; State = SessionState.Done;
            });
        }
        catch (Exception ex)
        {
            Apply(() => { CurrentRowId = null; State = SessionState.Failed; Error = ex.Message; });
        }
        finally { End(); }
    }

    public async Task UndoAsync()
    {
        if (!CanUndo || !TryBegin()) return;
        var runId = LastRun!.RunId;
        try
        {
            var results = await Task.Run(() => _backend.Undo(runId)).ConfigureAwait(false);
            var latest = await Task.Run(_backend.LatestRun).ConfigureAwait(false);
            Apply(() => { UndoResults = results; LastRun = latest; RanThisSession = false; });
        }
        catch (Exception ex)
        {
            Apply(() => Error = ex.Message);
        }
        finally { End(); }
        await LookAsync().ConfigureAwait(false);
    }

    /// Fills in the after-restart numbers once they exist.
    public async Task RefreshReceiptAsync()
    {
        if (LastRun is not { } run) return;
        var receipt = await Task.Run(() => _backend.RefreshReceipt(run.RunId)).ConfigureAwait(false);
        Apply(() => Receipt = receipt);
    }
}
